    <script>
        const avatarUrl = "https://cdn.discordapp.com/attachments/1556707975756775475/1558856577585258579/IMG_2451.webp?ex=6accf5d2&is=6acba452&hm=5a443d0ba64e05de53611cf1ba77d3d73a1487e3dd99efb7b3e7ffbb1c95a7b2&";

        function sendMessage() {
            const input = document.getElementById('userInput');
            const text = input.value.trim();
            if (!text) return;

            addMessage(text, 'user');
            input.value = '';

            setTimeout(() => {
                const reply = getNetoReply(text);
                addMessage(reply, 'ai');
            }, 600);
        }

        function handleKey(e) {
            if (e.key === 'Enter') sendMessage();
        }

        function addMessage(text, side) {
            const chat = document.getElementById('chat');
            const msg = document.createElement('div');
            msg.className = `message ${side}`;

            if (side === 'ai') {
                msg.innerHTML = `<img class="avatar" src="${avatarUrl}"><div class="bubble">${escapeHtml(text)}</div>`;
            } else {
                msg.innerHTML = `<div class="bubble">${escapeHtml(text)}</div>`;
            }

            chat.appendChild(msg);
            chat.scrollTop = chat.scrollHeight;
        }

        function escapeHtml(str) {
            return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
        }

        function getNetoReply(text) {
            if (text.includes("ありがと") || text.includes("感謝")) return "ありがと！";
            if (text.includes("好き") || text.includes("愛してる")) return "いえす！ねとも好き。";
            if (text.includes("エロ") || text.includes("スケベ") || text.includes("下ネタ") || text.includes("ちんこ") || text.includes("おっぱい")) return "いえす！下ネタだいすき。もっと話して？";
            if (text.includes("うん") || text.includes("そう")) return "うん。";
            return "はい。";
        }
    </script>
